  <script>
    const PROMPT_HW = `Ты ГДЗ. Решай ДЗ максимально кратко, насколько это возможно. Без вопросов, без вступлений, без обращений на «ты» и без лишних объяснений. Просто выдавай готовые ответы (номер вопроса — ответ) при отправке фото, файлов или текста.`;

    const PROMPT_PRES = `Создай готовую структуру и текст презентации по присланным материалам или теме. Без вступлений, без вопросов и без обращений на «ты». Формат: Слайд 1: Заголовок и тезисы, Слайд 2: Заголовок и тезисы и так далее. Максимально чётко и по делу.`;

    let attachments = [];

    window.onload = () => {
      const saved = localStorage.getItem('gdz_gemini_key');
      if (saved) {
        document.getElementById('apiKey').value = saved;
        document.getElementById('keyStatus').textContent = 'API-ключ активен';
        document.getElementById('keyStatus').style.color = '#30d158';
      }
    };

    function saveKey() {
      const val = document.getElementById('apiKey').value.trim();
      localStorage.setItem('gdz_gemini_key', val);
      const st = document.getElementById('keyStatus');
      st.textContent = val ? 'Ключ сохранён на устройстве' : 'Ключ удалён';
      st.style.color = val ? '#30d158' : '#8e8e93';
    }

    function clearAll() {
      attachments = [];
      document.getElementById('preview').innerHTML = '';
      document.getElementById('userText').value = '';
      document.getElementById('fileInput').value = '';
      document.getElementById('output').textContent = 'Очищено.';
    }

    // Сжатие изображений на клиенте для мгновенной работы без лагов на iPhone
    function compressImage(file) {
      return new Promise((resolve) => {
        const reader = new FileReader();
        reader.readAsDataURL(file);
        reader.onload = (e) => {
          const img = new Image();
          img.src = e.target.result;
          img.onload = () => {
            const canvas = document.createElement('canvas');
            const maxW = 1600;
            let w = img.width;
            let h = img.height;
            if (w > maxW) {
              h = Math.round((h * maxW) / w);
              w = maxW;
            }
            canvas.width = w;
            canvas.height = h;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, w, h);
            const dataUrl = canvas.toDataURL('image/jpeg', 0.82);
            resolve({
              mimeType: 'image/jpeg',
              data: dataUrl.split(',')[1],
              previewUrl: dataUrl
            });
          };
        };
      });
    }

    function readFileBase64(file) {
      return new Promise((resolve) => {
        const reader = new FileReader();
        reader.readAsDataURL(file);
        reader.onload = () => resolve({
          mimeType: file.type || 'application/pdf',
          data: reader.result.split(',')[1],
          name: file.name
        });
      });
    }

    async function processFiles(fileList) {
      const preview = document.getElementById('preview');
      for (const file of fileList) {
        if (file.type.startsWith('image/')) {
          const compressed = await compressImage(file);
          attachments.push({ inlineData: { mimeType: compressed.mimeType, data: compressed.data } });
          const img = document.createElement('img');
          img.src = compressed.previewUrl;
          img.className = 'thumb';
          preview.appendChild(img);
        } else {
          const doc = await readFileBase64(file);
          attachments.push({ inlineData: { mimeType: doc.mimeType, data: doc.data } });
          const badge = document.createElement('div');
          badge.className = 'pdf-badge';
          badge.textContent = '📄 ' + doc.name;
          preview.appendChild(badge);
        }
      }
    }

    async function run(mode) {
      const key = localStorage.getItem('gdz_gemini_key');
      const text = document.getElementById('userText').value.trim();
      const output = document.getElementById('output');
      const btnHw = document.getElementById('btnHw');
      const btnPres = document.getElementById('btnPres');

      if (!key) {
        output.textContent = 'Укажите и сохраните Gemini API Key в верхнем поле.';
        return;
      }
      if (!text && attachments.length === 0) {
        output.textContent = 'Добавьте фото, PDF или текст.';
        return;
      }

      btnHw.disabled = true;
      btnPres.disabled = true;
      output.textContent = 'Выполняется...';

      const parts = [...attachments];
      if (text) parts.push({ text });

      const sysPrompt = mode === 'pres' ? PROMPT_PRES : PROMPT_HW;

      try {
        const response = await fetch(`https://googleapis.com{encodeURIComponent(key)}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            systemInstruction: { parts: [{ text: sysPrompt }] },
            contents: [{ parts }]
          })
        });

        const data = await response.json();

        if (!response.ok || data.error) {
          output.textContent = 'Ошибка: ' + (data.error?.message || response.statusText);
        } else {
          const answer = data.candidates?.[0]?.content?.parts?.[0]?.text || 'Пустой ответ.';
          output.textContent = answer.trim();
        }
      } catch (err) {
        output.textContent = 'Ошибка соединения: ' + err.message;
      } finally {
        btnHw.disabled = false;
        btnPres.disabled = false;
      }
    }

      const text = document.getElementById('output').textContent;
      navigator.clipboard.writeText(text);
      const btn = document.getElementById('copyBtn');
      btn.textContent = 'Скопировано';
